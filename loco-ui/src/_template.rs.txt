use maud::{Markup, Render, html};

use crate::Ui;
use crate::props::{Prop, PropKind};

// TODO(5): every builder is `Clone` and `Debug` (`every_builder_is_clone_and_debug`), and its
// doc names its setters under **Setters.**: the tests find builders by that line, so keep the
// `///` lines and `#[derive]` directly above `pub struct` (no `//` line between them).
/// A widget, made by [`Ui::widget`].
///
/// **Setters.** Values: `.id(..)`; switches: `.quiet()`.
#[derive(Clone, Debug)]
pub struct Widget<'a> {
    label: &'a str,
    id: Option<&'a str>,
    quiet: bool,
}

impl Widget<'_> {
    // TODO(5, KEYS): the keys it answers to, from the elements it renders (a `<dialog>` closes
    // on Escape, radios move on arrows); `.script()` on what only enhance.rs does. Empty only
    // when it renders no control of its own (`builders_with_controls_list_their_keys`).
    /// The keys it answers to, listed by [`crate::props()`] and shown on its docs page.
    pub const KEYS: &'static [crate::props::Key] = &[
        crate::props::Key::new("Enter / Space", "Opens the widget."),
    ];

    // TODO(5, PROPS): one `Prop` per public setter, in the order written below
    // (`every_setter_is_in_props`, `props_follow_the_conventions`). Kinds: `Switch` (no
    // argument, switches on), `Condition` (takes a bool from a route), `Value`, `Number`, `Item` (an
    // adder), `Modifier` (changes the item added last). `.attr("x")` only when the setter sets
    // the HTML attribute `x`, and then the setter is named after it.
    /// Every setter with its kind, arguments, default and the HTML attribute it sets; listed by
    /// [`crate::props()`] and kept in step with the setters by a test.
    pub const PROPS: &'static [Prop] = &[
        Prop::new("id", PropKind::Value, "id: &'a str")
            .attr("id")
            .doc("Replace the id derived from the label."),
        Prop::new("quiet", PropKind::Switch, "").doc("A lighter look."),
    ];
}

impl Ui {
    // TODO(5, constructor): required arguments only, text first; everything else a setter.
    // Read query and cookie state here from `self.param(..)` / `self.state()`, not from an
    // argument. Keep `&self.caps` (or what you need of `self`) in the builder if it branches.
    /// A widget reading `label`.
    pub fn widget<'a>(&self, label: &'a str) -> Widget<'a> {
        Widget {
            label,
            id: None,
            quiet: false,
        }
    }
}

impl<'a> Widget<'a> {
    /// Replace the id derived from the label.
    pub fn id(mut self, id: &'a str) -> Self {
        self.id = Some(id);
        self
    }

    /// A lighter look.
    pub fn quiet(mut self) -> Self {
        self.quiet = true;
        self
    }
}

impl Render for Widget<'_> {
    fn render(&self) -> Markup {
        // TODO(5, ids): an id the caller does not care about is `lui-<name>-` and
        // `crate::slug(label)`.
        let _id = self
            .id
            .map_or_else(|| format!("lui-widget-{}", crate::slug(self.label)), str::to_string);
        // TODO(4, classes): one `lui-widget` class on the root, `lui-widget-<part>` on parts.
        // TODO(8, primitives): buttons, inputs, badges, cards, icons and layouts come from
        // `ui.button`, `ui.input(..).hide_label()`, `ui.badge`, `ui.card`, `Icon`, `ui.stack`..
        // never a raw `<button>` or visible `<input>`.
        // TODO(5, caps): `if self.caps.has(Cap::X) { modern } else { fallback }`, one emitted.
        // TODO(5, swap): a root that updates in place gets
        // `id=(crate::enhance::swap_id("widget", key)) data-lui="swap"`, same markup without script.
        // TODO(i18n): no English literal here; words come from `ui.text(Text::..)`
        // (`components_write_no_english_outside_the_string_table`).
        let class = if self.quiet {
            "lui-widget lui-widget-quiet"
        } else {
            "lui-widget"
        };
        html! { div class=(class) { (self.label) } }
    }
}
